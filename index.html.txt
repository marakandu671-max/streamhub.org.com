                <button onclick="filterCategory('Nature')" class="category-btn bg-slate-900 hover:bg-slate-800 text-slate-300 px-4 py-2 rounded-full text-sm font-semibold transition whitespace-nowrap border border-slate-800">Nature</button>

    <!-- App Logic -->
    <script>
        const sampleVideos = [
            { id: 'jfKfPfyJRdk', title: 'Lofi Girl - lofi hip hop radio 📚 - beats to relax/study to', category: 'Music', duration: 'LIVE' },
            { id: 'aqz-KE-bpKQ', title: 'Big Buck Bunny (Blender Foundation Open Movie)', category: 'Trailers', duration: '9:56' },
            { id: 'LXb3EKWsInQ', title: 'Costa Rica In 4K 60fps (Ultra HD)', category: 'Nature', duration: '5:59' },
            { id: '5qap5aO4i9A', title: 'Synthwave Radio - Chill synth / retro beats', category: 'Music', duration: 'LIVE' },
            { id: '1LA4QzGeaaQ', title: 'Sintel - Official Blender Open Movie', category: 'Trailers', duration: '14:48' }
        ];

        let currentCategory = 'All';

        function renderVideos(videosToRender) {
            const grid = document.getElementById('video-grid');
            if (videosToRender.length === 0) {
                grid.innerHTML = `<div class="col-span-full text-center py-12 text-slate-500">No videos found.</div>`;
                return;
            }

            grid.innerHTML = videosToRender.map(video => `
                <div onclick="openVideoModal('${video.id}', '${video.title.replace(/'/g, "\\'")}')" class="bg-slate-900 border border-slate-800 rounded-2xl overflow-hidden group cursor-pointer hover:border-slate-700 transition duration-300 flex flex-col">
                    <div class="relative aspect-video bg-slate-950 overflow-hidden">
                        <img src="https://img.youtube.com/vi/${video.id}/hqdefault.jpg" alt="${video.title}" class="w-full h-full object-cover group-hover:scale-105 transition duration-500">
                        <div class="absolute inset-0 bg-black/30 group-hover:bg-black/10 transition flex items-center justify-center">
                            <div class="w-12 h-12 rounded-full bg-red-600/90 text-white flex items-center justify-center shadow-lg transform scale-90 group-hover:scale-110 transition">
                                <i class="fa-solid fa-play ml-0.5"></i>
                            </div>
                        </div>
                        <span class="absolute bottom-2 right-2 bg-black/80 text-xs px-2 py-1 rounded font-medium">${video.duration}</span>
                    </div>
                    <div class="p-4 flex-1 flex flex-col justify-between">
                        <h3 class="font-semibold text-sm line-clamp-2 group-hover:text-red-400 transition">${video.title}</h3>
                        <span class="text-xs text-slate-500 mt-2">${video.category}</span>
                    </div>
                </div>
            `).join('');
        }

        function filterCategory(category) {
            currentCategory = category;
            document.querySelectorAll('.category-btn').forEach(btn => {
                if (btn.innerText === category) {
                    btn.className = 'category-btn bg-red-600 text-white px-4 py-2 rounded-full text-sm font-semibold transition whitespace-nowrap';
                } else {
                    btn.className = 'category-btn bg-slate-900 hover:bg-slate-800 text-slate-300 px-4 py-2 rounded-full text-sm font-semibold transition whitespace-nowrap border border-slate-800';
                }
            });

            if (category === 'All') {
                renderVideos(sampleVideos);
            } else {
                renderVideos(sampleVideos.filter(v => v.category === category));
            }
        }

        function searchVideos(query) {
            const filtered = sampleVideos.filter(v => 
                (currentCategory === 'All' || v.category === currentCategory) &&
                v.title.toLowerCase().includes(query.toLowerCase())
            );
            renderVideos(filtered);
        }

        function openVideoModal(id, title) {
            document.getElementById('modal-title').innerText = title;
            document.getElementById('modal-iframe').src = `https://www.youtube.com/embed/${id}?autoplay=1`;
            document.getElementById('video-modal').classList.remove('hidden');
        }

        function closeVideoModal() {
            document.getElementById('modal-iframe').src = '';
            document.getElementById('video-modal').classList.add('hidden');
        }

        function openCustomVideoModal() {
            document.getElementById('custom-url-modal').classList.remove('hidden');
        }

        function closeCustomVideoModal() {
            document.getElementById('custom-url-modal').classList.add('hidden');
        }

        function extractVideoId(urlOrId) {
            if (!urlOrId) return '';
            if (urlOrId.length === 11) return urlOrId;
            const regExp = /^.*(youtu.be\/|v\/|u\/\w\/|embed\/|watch\?v=|\&v=)([^#\&\?]*).*/;
            const match = urlOrId.match(regExp);
            return (match && match[2].length === 11) ? match[2] : null;
        }

        function playCustomUrl() {
            const input = document.getElementById('custom-url-input').value.trim();
            const id = extractVideoId(input);
            if (id) {
                closeCustomVideoModal();
                openVideoModal(id, 'Custom Stream');
                document.getElementById('custom-url-input').value = '';
            } else {
                alert('Please enter a valid YouTube link or ID.');
            }
        }

        // Initial Load
        renderVideos(sampleVideos);
    </script>
